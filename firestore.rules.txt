rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function signedIn(){return request.auth != null;}

    match /users/{userId}/{document=**} {
      allow read, write: if signedIn() && request.auth.uid == userId;
    }

    match /publicProfiles/{userId} {
      allow read: if signedIn();
      allow create, update, delete: if signedIn() && request.auth.uid == userId;
    }

    match /friendRequests/{id} {
      allow create: if signedIn() && request.resource.data.fromUid == request.auth.uid;
      allow read: if signedIn() && (resource.data.fromUid == request.auth.uid || resource.data.toUid == request.auth.uid);
      allow update, delete: if signedIn() && (resource.data.fromUid == request.auth.uid || resource.data.toUid == request.auth.uid);
    }

    match /friendships/{id} {
      allow create: if signedIn() && request.auth.uid in request.resource.data.members;
      allow read, delete: if signedIn() && request.auth.uid in resource.data.members;
      allow update: if false;
    }

    match /feedPosts/{postId} {
      allow create: if signedIn() && request.resource.data.ownerUid == request.auth.uid;
      allow read: if signedIn() && request.auth.uid in resource.data.viewers;
      allow update, delete: if signedIn() && resource.data.ownerUid == request.auth.uid;

      match /likes/{userId} {
        allow read: if signedIn() && request.auth.uid in get(/databases/$(database)/documents/feedPosts/$(postId)).data.viewers;
        allow create, update, delete: if signedIn() && request.auth.uid == userId;
      }

      match /comments/{commentId} {
        allow read: if signedIn() && request.auth.uid in get(/databases/$(database)/documents/feedPosts/$(postId)).data.viewers;
        allow create: if signedIn() && request.resource.data.uid == request.auth.uid;
        allow delete: if signedIn() && resource.data.uid == request.auth.uid;
        allow update: if false;
      }
    }
  }
}
